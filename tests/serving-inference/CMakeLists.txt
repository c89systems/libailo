cmake_minimum_required(VERSION 3.24)
project(ailo_serving_acceptance NONE)
enable_testing()
foreach(input AILO_FIXTURE AILO_MODEL AILO_EVIDENCE)
  if(NOT IS_ABSOLUTE "${${input}}" OR NOT EXISTS "${${input}}")
    message(FATAL_ERROR "${input} must name an existing absolute path")
  endif()
endforeach()
foreach(scenario protocol compaction transport)
  add_test(NAME trained-serving-${scenario} COMMAND bash "${CMAKE_CURRENT_LIST_DIR}/../test_serving_inference.sh" "${AILO_FIXTURE}" "${AILO_MODEL}" "${AILO_EVIDENCE}" "${scenario}")
  set_tests_properties(trained-serving-${scenario} PROPERTIES TIMEOUT 180 LABELS "e2e;inference;cpu")
endforeach()
add_test(NAME trained-serving-codex COMMAND bash "${CMAKE_CURRENT_LIST_DIR}/../test_serving_agent_inference.sh" "${AILO_FIXTURE}" "${AILO_MODEL}" "${AILO_EVIDENCE}")
set_tests_properties(trained-serving-codex PROPERTIES TIMEOUT 300 LABELS "e2e;inference;cpu;agent")
