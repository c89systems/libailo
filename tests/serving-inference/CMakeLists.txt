cmake_minimum_required(VERSION 3.24)
project(ailo_serving_acceptance NONE)
enable_testing()
foreach(input AILO_FIXTURE AILO_MODEL AILO_EVIDENCE AILO_VISION_MODEL AILO_VISION_PROJECTOR AILO_AUDIO_MODEL AILO_AUDIO_PROJECTOR AILO_AUDIO_SAMPLE)
  if(NOT IS_ABSOLUTE "${${input}}" OR NOT EXISTS "${${input}}")
    message(FATAL_ERROR "${input} must name an existing absolute path")
  endif()
endforeach()
if(NOT AILO_BACKEND MATCHES "^(cpu|vulkan)$")
  message(FATAL_ERROR "AILO_BACKEND must be cpu or vulkan")
endif()
set(environment "AILO_TEST_BACKEND=${AILO_BACKEND};AILO_TEST_TRACE=${AILO_TRACE};MESA_SHADER_CACHE_DISABLE=true;VK_ICD_FILENAMES=/ailo-forbidden-icd.json;VK_DRIVER_FILES=/ailo-forbidden-icd.json")
foreach(scenario protocol compaction transport)
  add_test(NAME trained-serving-${scenario} COMMAND bash "${CMAKE_CURRENT_LIST_DIR}/../test_serving_inference.sh" "${AILO_FIXTURE}" "${AILO_MODEL}" "${AILO_EVIDENCE}" "${scenario}")
  set_tests_properties(trained-serving-${scenario} PROPERTIES TIMEOUT 180 LABELS "e2e;inference;${AILO_BACKEND}" ENVIRONMENT "${environment};AILO_TEST_KIND=text")
endforeach()
add_test(NAME trained-serving-codex COMMAND bash "${CMAKE_CURRENT_LIST_DIR}/../test_serving_agent_inference.sh" "${AILO_FIXTURE}" "${AILO_MODEL}" "${AILO_EVIDENCE}")
set_tests_properties(trained-serving-codex PROPERTIES TIMEOUT 300 LABELS "e2e;inference;${AILO_BACKEND};agent" ENVIRONMENT "${environment};AILO_TEST_KIND=text")
foreach(kind vision audio)
  if(kind STREQUAL vision)
    set(model "${AILO_VISION_MODEL}")
    set(projector "${AILO_VISION_PROJECTOR}")
    set(sample "${CMAKE_CURRENT_LIST_DIR}/../fixtures/media/red.jpg")
  else()
    set(model "${AILO_AUDIO_MODEL}")
    set(projector "${AILO_AUDIO_PROJECTOR}")
    set(sample "${AILO_AUDIO_SAMPLE}")
  endif()
  add_test(NAME trained-serving-${kind} COMMAND bash "${CMAKE_CURRENT_LIST_DIR}/../test_serving_media.sh" "${AILO_FIXTURE}" "${model}" "${AILO_EVIDENCE}" "${kind}" "${sample}")
  set_tests_properties(trained-serving-${kind} PROPERTIES TIMEOUT 240 LABELS "e2e;inference;${AILO_BACKEND};media" ENVIRONMENT "${environment};AILO_TEST_KIND=${kind};AILO_TEST_PROJECTOR=${projector}")
endforeach()
