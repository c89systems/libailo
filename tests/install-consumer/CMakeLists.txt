cmake_minimum_required(VERSION 3.24)
project(ailo_downstream LANGUAGES C)
if(AILO_CONSUMER_CXX)
  enable_language(CXX)
  configure_file(consumer.c consumer.cpp COPYONLY)
  set(source "${CMAKE_CURRENT_BINARY_DIR}/consumer.cpp")
else()
  set(source consumer.c)
endif()
if(AILO_CONSUMER_KIND MATCHES "^pkg-")
  find_package(PkgConfig REQUIRED)
  if(AILO_CONSUMER_KIND STREQUAL "pkg-static")
    set(pkg_arguments --static ailo-static)
  else()
    set(pkg_arguments ailo)
  endif()
  execute_process(COMMAND "${PKG_CONFIG_EXECUTABLE}" --libs ${pkg_arguments} OUTPUT_VARIABLE flags COMMAND_ERROR_IS_FATAL ANY OUTPUT_STRIP_TRAILING_WHITESPACE)
  separate_arguments(library UNIX_COMMAND "${flags}")
  set(pkg_libraries "")
  set(pkg_library_directories "")
  foreach(flag IN LISTS library)
    if(flag MATCHES "^-L(.+)$")
      list(APPEND pkg_library_directories "${CMAKE_MATCH_1}")
    else()
      list(APPEND pkg_libraries "${flag}")
    endif()
  endforeach()
  set(library "${pkg_libraries}")
  execute_process(COMMAND "${PKG_CONFIG_EXECUTABLE}" --cflags ${pkg_arguments} OUTPUT_VARIABLE flags COMMAND_ERROR_IS_FATAL ANY OUTPUT_STRIP_TRAILING_WHITESPACE)
  separate_arguments(pkg_cflags UNIX_COMMAND "${flags}")
  execute_process(COMMAND "${PKG_CONFIG_EXECUTABLE}" --modversion ${pkg_arguments} OUTPUT_VARIABLE expected_version COMMAND_ERROR_IS_FATAL ANY OUTPUT_STRIP_TRAILING_WHITESPACE)
else()
  find_package(ailo CONFIG REQUIRED COMPONENTS "${AILO_CONSUMER_KIND}")
  set(library "ailo::${AILO_CONSUMER_KIND}")
  set(expected_version "${ailo_VERSION}")
endif()
configure_file("${AILO_POLICY_ROOT}/tests/model.c" "${CMAKE_CURRENT_BINARY_DIR}/model.c" COPYONLY)
if(AILO_CONSUMER_CXX)
  configure_file("${CMAKE_CURRENT_BINARY_DIR}/model.c" "${CMAKE_CURRENT_BINARY_DIR}/model.cpp" COPYONLY)
  set(model_source "${CMAKE_CURRENT_BINARY_DIR}/model.cpp")
else()
  set(model_source "${CMAKE_CURRENT_BINARY_DIR}/model.c")
endif()
add_executable(consumer ${source})
add_executable(model_consumer ${model_source})
# Only verification executables receive local toolchain/SDK runtime paths.
# These settings never become part of installed package metadata.
set(AILO_DEPENDENCY_ROOT "${AILO_TEST_DEPENDENCY_ROOT}")
include("${AILO_POLICY_ROOT}/cmake/AiloPolicy.cmake")
foreach(target consumer model_consumer)
  target_link_libraries(${target} PRIVATE ${library})
  target_link_directories(${target} PRIVATE ${pkg_library_directories})
  target_compile_options(${target} PRIVATE ${pkg_cflags} -Wall -Wextra -Wpedantic -Werror "$<$<COMPILE_LANGUAGE:C>:-std=c89;-pedantic-errors>")
  target_link_options(${target} PRIVATE -Wl,--fatal-warnings "-Wl,-rpath-link,${AILO_TEST_PREFIX}/lib/ailo/native")
  ailo_local_executable(${target})
  if(AILO_TARGET_ID MATCHES "-musl$" AND AILO_CONSUMER_KIND MATCHES "^(pkg-)?static$")
    target_link_options(${target} PRIVATE -static)
  endif()
  set_property(TARGET ${target} APPEND PROPERTY BUILD_RPATH "${AILO_TEST_PREFIX}/lib" "${AILO_TEST_PREFIX}/lib/ailo/native")
endforeach()
target_compile_definitions(consumer PRIVATE "EXPECTED_VERSION=\"${expected_version}\"")
file(READ "${AILO_POLICY_ROOT}/cmake/dependencies.json" pins)
file(READ "${AILO_POLICY_ROOT}/docs/upstream-api.json" inventory)
string(JSON native_version GET "${pins}" llama version)
string(REGEX REPLACE "^v" "" native_version "${native_version}")
string(JSON native_revision GET "${inventory}" revision)
target_compile_definitions(consumer PRIVATE
  "EXPECTED_LLAMA_VERSION=\"${native_version}\""
  "EXPECTED_LLAMA_REVISION=\"${native_revision}\"")
file(GLOB headers "${AILO_TEST_PREFIX}/include/ailo/*.h")
foreach(header IN LISTS headers)
  get_filename_component(name "${header}" NAME)
  string(MAKE_C_IDENTIFIER "${name}" tag)
  if(AILO_CONSUMER_CXX)
    set(extension cpp)
  else()
    set(extension c)
  endif()
  set(probe "${CMAKE_CURRENT_BINARY_DIR}/header_${tag}.${extension}")
  file(WRITE "${probe}" "#include <ailo/${name}>\nint probe_${tag}(void) { return 0; }\n")
  add_library(header_${tag} OBJECT "${probe}")
  target_include_directories(header_${tag} PRIVATE "${AILO_TEST_PREFIX}/include")
  target_compile_options(header_${tag} PRIVATE -Wall -Wextra -Wpedantic -Werror "$<$<COMPILE_LANGUAGE:C>:-std=c89;-pedantic-errors>")
endforeach()
