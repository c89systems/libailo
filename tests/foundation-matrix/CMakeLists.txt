cmake_minimum_required(VERSION 3.24)
project(ailo_foundation_matrix LANGUAGES C CXX)
enable_testing()
set(AILO_DEPENDENCY_ROOT "${AILO_PROJECT_ROOT}/.cache/deps/${AILO_TARGET_ID}")
set(contracts "")
foreach(component cpkt cpktmisc lonejson pslog liblql lockdc)
  list(APPEND CMAKE_PREFIX_PATH "${AILO_DEPENDENCY_ROOT}/${component}/install")
  list(APPEND contracts "${AILO_PROJECT_ROOT}/.cache/dependency-contracts/${AILO_TARGET_ID}/${component}.txt")
endforeach()
find_package(CpktCore 0.6.1 EXACT CONFIG REQUIRED)
find_package(CpktMisc CONFIG REQUIRED)
find_package(CpktFfi 3.8.0 EXACT CONFIG REQUIRED)
find_package(lonejson 0.45.0 EXACT CONFIG REQUIRED)
find_package(pslog 0.13.0 EXACT CONFIG REQUIRED)
find_package(liblql 0.4.1 EXACT CONFIG REQUIRED)
find_package(lockdc 0.27.0 EXACT CONFIG REQUIRED COMPONENTS static shared)
if(NOT CpktCore_TARGET_ID STREQUAL AILO_TARGET_ID OR NOT CpktMisc_VERSION VERSION_EQUAL "0.3.0")
  message(FATAL_ERROR "Incorrect foundation SDK identity")
endif()
include("${AILO_PROJECT_ROOT}/cmake/AiloPolicy.cmake")
if(CMAKE_CROSSCOMPILING)
  find_program(runner qemu-aarch64 REQUIRED)
endif()
foreach(kind static shared)
  add_executable(ffi_${kind} "${AILO_PROJECT_ROOT}/tests/dependency_ffi.cpp")
  target_link_libraries(ffi_${kind} PRIVATE cpkt::libffi_${kind})
  target_compile_features(ffi_${kind} PRIVATE cxx_std_17)
  target_compile_options(ffi_${kind} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(ffi_${kind} PRIVATE -static-libstdc++ -static-libgcc)
  ailo_local_executable(ffi_${kind})
  if(CMAKE_CROSSCOMPILING)
    set_property(TARGET ffi_${kind} PROPERTY CROSSCOMPILING_EMULATOR "${runner};-L;${CMAKE_SYSROOT}")
  endif()
  add_test(NAME ffi-${kind} COMMAND ffi_${kind})
  set_tests_properties(ffi-${kind} PROPERTIES TIMEOUT 15)
  add_executable(collections_${kind} "${AILO_PROJECT_ROOT}/tests/lockdc_collections.cpp")
  target_link_libraries(collections_${kind} PRIVATE lockdc::${kind})
  target_compile_features(collections_${kind} PRIVATE cxx_std_17)
  target_compile_options(collections_${kind} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(collections_${kind} PRIVATE -static-libstdc++ -static-libgcc)
  if(AILO_TARGET_ID MATCHES "-musl$" AND kind STREQUAL "static")
    target_link_options(collections_${kind} PRIVATE -static)
  endif()
  ailo_local_executable(collections_${kind})
  if(CMAKE_CROSSCOMPILING)
    set_property(TARGET collections_${kind} PROPERTY CROSSCOMPILING_EMULATOR "${runner};-L;${CMAKE_SYSROOT}")
  endif()
  add_test(NAME collections-${kind} COMMAND collections_${kind}
    "${AILO_PROJECT_ROOT}/build/foundation-matrix/lockdc-collections-${AILO_TARGET_ID}-${kind}")
  set_tests_properties(collections-${kind} PROPERTIES TIMEOUT 60)
endforeach()
include("${AILO_PROJECT_ROOT}/cmake/AiloDependencyInputs.cmake")
ailo_track_dependency_inputs(${contracts})
