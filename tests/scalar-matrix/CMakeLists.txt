cmake_minimum_required(VERSION 3.24)
project(ailo_scalar_matrix LANGUAGES C CXX)
if(NOT EXISTS "${AILO_PROJECT_ROOT}/include/ailo/base.h")
  message(FATAL_ERROR "AILO_PROJECT_ROOT must identify the libailo repository")
endif()
include("${AILO_PROJECT_ROOT}/cmake/AiloPolicy.cmake")
enable_testing()
add_library(ailo_scalars STATIC "${AILO_PROJECT_ROOT}/src/ailo_error.c" "${AILO_PROJECT_ROOT}/src/ailo_integer.c")
target_include_directories(ailo_scalars PUBLIC "${AILO_PROJECT_ROOT}/include")
ailo_c89_target(ailo_scalars)
add_executable(ailo_values "${AILO_PROJECT_ROOT}/tests/values.c")
target_link_libraries(ailo_values PRIVATE ailo_scalars)
ailo_c89_target(ailo_values)
ailo_local_executable(ailo_values)
target_link_options(ailo_values PRIVATE -Wl,--fatal-warnings)
add_executable(ailo_integer_bridge "${AILO_PROJECT_ROOT}/tests/integer_bridge.cpp")
target_include_directories(ailo_integer_bridge PRIVATE "${AILO_PROJECT_ROOT}/src")
target_link_libraries(ailo_integer_bridge PRIVATE ailo_scalars)
target_compile_options(ailo_integer_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_options(ailo_integer_bridge PRIVATE -Wl,--fatal-warnings)
target_compile_features(ailo_integer_bridge PRIVATE cxx_std_17)
ailo_local_executable(ailo_integer_bridge)
if(CMAKE_CROSSCOMPILING)
  if(CMAKE_SYSTEM_PROCESSOR STREQUAL "arm")
    find_program(runner qemu-arm REQUIRED)
  else()
    find_program(runner qemu-aarch64 REQUIRED)
  endif()
  foreach(target ailo_values ailo_integer_bridge)
    set_property(TARGET ${target} PROPERTY CROSSCOMPILING_EMULATOR "${runner};-L;${CMAKE_SYSROOT}")
  endforeach()
endif()
add_test(NAME values COMMAND ailo_values)
add_test(NAME native-bridge COMMAND ailo_integer_bridge)
set_tests_properties(values native-bridge PROPERTIES TIMEOUT 30)
