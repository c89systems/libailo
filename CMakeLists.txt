cmake_minimum_required(VERSION 3.24)
if(NOT CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_SOURCE_DIR}/cmake/AiloToolchain.cmake" CACHE FILEPATH "Pinned collection")
endif()
project(ailo VERSION 0.0.0 LANGUAGES C CXX ASM)
include(cmake/AiloPolicy.cmake)
include(CTest)
find_package(Threads REQUIRED)
option(AILO_VULKAN "Build required Vulkan acceleration plus CPU" ON)
option(AILO_BUILD_TESTS "Build deterministic tests" ON)
option(AILO_BUILD_EXAMPLES "Build examples" ON)
option(AILO_BUILD_STATIC "Build the static project library" ON)
option(AILO_BUILD_SHARED "Build the shared project library" ON)
if(AILO_BUILD_TESTS)
  # Python is confined to the compiler AST inventory and test HTTP fixtures.
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
endif()

set(AILO_BASE_SOURCES src/ailo_error.c src/ailo_integer.c)
if(AILO_BUILD_STATIC)
  add_library(ailo_static STATIC ${AILO_BASE_SOURCES})
  add_library(ailo::static ALIAS ailo_static)
  set_target_properties(ailo_static PROPERTIES OUTPUT_NAME ailo)
  target_include_directories(ailo_static PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
  ailo_c89_target(ailo_static)
endif()
if(AILO_BUILD_SHARED)
  add_library(ailo_shared SHARED ${AILO_BASE_SOURCES})
  add_library(ailo::shared ALIAS ailo_shared)
  set_target_properties(ailo_shared PROPERTIES OUTPUT_NAME ailo VERSION ${PROJECT_VERSION} SOVERSION 0)
  target_include_directories(ailo_shared PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
  ailo_c89_target(ailo_shared)
  file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/ailo.exports" exports)
  set(export_map "{ global:\n")
  foreach(symbol IN LISTS exports)
    if(NOT symbol MATCHES "^ailo_[A-Za-z0-9_]+$")
      message(FATAL_ERROR "Invalid public symbol in ailo.exports: ${symbol}")
    endif()
    string(APPEND export_map "  ${symbol};\n")
  endforeach()
  string(APPEND export_map "local: *; };\n")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ailo.map" "${export_map}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/ailo.exports")
  set_property(TARGET ailo_shared APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/ailo.map")
  target_link_options(ailo_shared PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_BINARY_DIR}/ailo.map" -Wl,--fatal-warnings)
endif()
if(AILO_BUILD_EXAMPLES)
  add_executable(ailo_scalars_example examples/scalars.c)
  if(AILO_BUILD_STATIC)
    target_link_libraries(ailo_scalars_example PRIVATE ailo_static)
  elseif(AILO_BUILD_SHARED)
    target_link_libraries(ailo_scalars_example PRIVATE ailo_shared)
  else()
    message(FATAL_ERROR "Examples need AILO_BUILD_STATIC or AILO_BUILD_SHARED")
  endif()
  ailo_c89_target(ailo_scalars_example)
  target_link_options(ailo_scalars_example PRIVATE -Wl,--fatal-warnings)
  ailo_local_executable(ailo_scalars_example)
endif()
set(AILO_DEPENDENCY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.cache/deps/${AILO_TARGET_ID}")
set(AILO_DEPENDENCY_BUILD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.cache/deps-build/${AILO_TARGET_ID}")
set(CURL_DIR "${AILO_DEPENDENCY_ROOT}/cpkt/install/lib/cmake/CURL")
set(lonejson_DIR "${AILO_DEPENDENCY_ROOT}/lonejson/install/lib/cmake/lonejson")
find_package(CURL CONFIG REQUIRED)
find_package(lonejson CONFIG REQUIRED)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build native dependencies as separate archives" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_NATIVE OFF CACHE BOOL "Portable CPU instructions" FORCE)
set(GGML_CPU ON CACHE BOOL "Required CPU backend" FORCE)
set(GGML_VULKAN ${AILO_VULKAN} CACHE BOOL "Vulkan backend" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "No ambient compiler cache" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "No separate OpenMP runtime" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "No C++ utility surface" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "Library only" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "Project-owned tests" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "Project-owned examples" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "Public multimodal API" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "No hidden subprocesses" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "Model acquisition belongs to libailo" FORCE)
set(AILO_SHADER_TOOLCHAIN "${CMAKE_CURRENT_SOURCE_DIR}/cmake/AiloShaderToolchain.cmake")
set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "${AILO_SHADER_TOOLCHAIN}" CACHE FILEPATH "Pinned native shader generator collection" FORCE)
if(AILO_VULKAN)
  set(Vulkan_INCLUDE_DIR "${AILO_DEPENDENCY_ROOT}/vulkan-headers/install/include" CACHE PATH "Pinned Vulkan headers" FORCE)
  set(Vulkan_LIBRARY "${AILO_DEPENDENCY_ROOT}/vulkan-loader/install/lib/libvulkan.so" CACHE FILEPATH "Pinned Vulkan loader" FORCE)
  set(SPIRV-Headers_DIR "${AILO_DEPENDENCY_ROOT}/spirv-headers/install/share/cmake/SPIRV-Headers")
  find_package(SPIRV-Headers CONFIG REQUIRED)
endif()
add_subdirectory("${AILO_DEPENDENCY_BUILD_ROOT}/llama/source" "${AILO_DEPENDENCY_BUILD_ROOT}/llama/build" EXCLUDE_FROM_ALL)
if(AILO_VULKAN)
  target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
endif()

if(AILO_BUILD_TESTS)
  if(AILO_BUILD_EXAMPLES)
    add_test(NAME example-scalars COMMAND ailo_scalars_example)
    add_test(NAME example-scalars-help COMMAND ailo_scalars_example --help)
    set_tests_properties(example-scalars example-scalars-help PROPERTIES TIMEOUT 10 LABELS "example-smoke;local")
  endif()
  if(AILO_BUILD_STATIC)
    add_executable(ailo_values_static tests/values.c)
    target_link_libraries(ailo_values_static PRIVATE ailo_static)
    ailo_c89_target(ailo_values_static)
    ailo_local_executable(ailo_values_static)
    add_test(NAME values-static COMMAND ailo_values_static)
    set_tests_properties(values-static PROPERTIES TIMEOUT 10 LABELS "unit;local")
    add_executable(ailo_integer_bridge tests/integer_bridge.cpp)
    target_include_directories(ailo_integer_bridge PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(ailo_integer_bridge PRIVATE ailo_static)
    target_compile_options(ailo_integer_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(ailo_integer_bridge PRIVATE -Wl,--fatal-warnings)
    target_compile_features(ailo_integer_bridge PRIVATE cxx_std_17)
    ailo_local_executable(ailo_integer_bridge)
    add_test(NAME integer-bridge COMMAND ailo_integer_bridge)
    set_tests_properties(integer-bridge PROPERTIES TIMEOUT 10 LABELS "unit;local")
  endif()
  if(AILO_BUILD_SHARED)
    add_executable(ailo_values_shared tests/values.c)
    target_link_libraries(ailo_values_shared PRIVATE ailo_shared)
    ailo_c89_target(ailo_values_shared)
    ailo_local_executable(ailo_values_shared)
    add_test(NAME values-shared COMMAND ailo_values_shared)
    set_tests_properties(values-shared PROPERTIES TIMEOUT 10 LABELS "unit;local")
    add_test(NAME shared-exports COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_exports.sh"
      --library "$<TARGET_FILE:ailo_shared>" --nm "${CMAKE_NM}" --cc "${CMAKE_C_COMPILER}"
      --sysroot "${CMAKE_SYSROOT}" --work "${CMAKE_CURRENT_BINARY_DIR}/shared-contract")
    set_tests_properties(shared-exports PROPERTIES TIMEOUT 30 LABELS "contract;local")
  endif()
  add_executable(ailo_backend_test tests/backend.cpp)
  target_compile_options(ailo_backend_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(ailo_backend_test PRIVATE -Wl,--fatal-warnings)
  target_link_libraries(ailo_backend_test PRIVATE llama ggml mtmd)
  if(AILO_VULKAN)
    target_compile_definitions(ailo_backend_test PRIVATE AILO_TEST_VULKAN)
  endif()
  ailo_local_executable(ailo_backend_test)
  add_test(NAME backend COMMAND ailo_backend_test)
  set_tests_properties(backend PROPERTIES TIMEOUT 60 LABELS "unit;local")
  add_executable(ailo_c89_policy tests/c89_policy.c)
  ailo_c89_target(ailo_c89_policy)
  ailo_local_executable(ailo_c89_policy)
  target_link_options(ailo_c89_policy PRIVATE -Wl,--fatal-warnings)
  add_test(NAME c89-policy COMMAND ailo_c89_policy)
  set_tests_properties(c89-policy PROPERTIES TIMEOUT 10 LABELS "unit;contract")
  add_test(NAME upstream-inventory COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/upstream_inventory.py" --source "${AILO_DEPENDENCY_BUILD_ROOT}/llama/source")
  set_tests_properties(upstream-inventory PROPERTIES TIMEOUT 60 LABELS "local;contract")
  add_test(NAME archive-cache COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_archive_cache.py")
  set_tests_properties(archive-cache PROPERTIES TIMEOUT 30 LABELS "unit;local;contract")
  add_test(NAME build-drivers COMMAND ${CMAKE_COMMAND} -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_build_drivers.cmake")
  set_tests_properties(build-drivers PROPERTIES TIMEOUT 30 LABELS "unit;local;contract")
  if(NOT CMAKE_CROSSCOMPILING AND AILO_TARGET_ID STREQUAL "x86_64-linux-gnu" AND AILO_VULKAN AND CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_test(NAME lifecycle COMMAND ${CMAKE_COMMAND} "-DAILO_BUILD_DIRECTORY=${CMAKE_CURRENT_BINARY_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_lifecycle.cmake")
    set_tests_properties(lifecycle PROPERTIES TIMEOUT 60 LABELS "local;contract")
  endif()
endif()
