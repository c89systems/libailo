cmake_minimum_required(VERSION 3.24)
if(NOT CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_SOURCE_DIR}/cmake/AiloToolchain.cmake" CACHE FILEPATH "Pinned collection")
endif()
project(ailo VERSION 0.0.0 LANGUAGES C CXX ASM)
include(cmake/AiloPolicy.cmake)
include(CTest)
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
option(AILO_VULKAN "Build required Vulkan acceleration plus CPU" ON)
option(AILO_BUILD_TESTS "Build deterministic tests" ON)
option(AILO_BUILD_EXAMPLES "Build examples" ON)
set(AILO_DEPENDENCY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.cache/deps/${AILO_TARGET_ID}")
set(AILO_DEPENDENCY_BUILD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.cache/deps-build/${AILO_TARGET_ID}")
set(CURL_DIR "${AILO_DEPENDENCY_ROOT}/cpkt/install/lib/cmake/CURL")
set(lonejson_DIR "${AILO_DEPENDENCY_ROOT}/lonejson/install/lib/cmake/lonejson")
find_package(CURL CONFIG REQUIRED)
find_package(lonejson CONFIG REQUIRED)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build native dependencies as separate archives" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_NATIVE OFF CACHE BOOL "Portable CPU instructions" FORCE)
set(GGML_CPU ON CACHE BOOL "Required CPU backend" FORCE)
set(GGML_VULKAN ${AILO_VULKAN} CACHE BOOL "Vulkan backend" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "No ambient compiler cache" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "No separate OpenMP runtime" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "No C++ utility surface" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "Library only" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "Project-owned tests" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "Project-owned examples" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "Public multimodal API" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "No hidden subprocesses" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "Model acquisition belongs to libailo" FORCE)
set(AILO_SHADER_TOOLCHAIN "${CMAKE_CURRENT_SOURCE_DIR}/cmake/AiloShaderToolchain.cmake")
if(AILO_VULKAN)
  set(Vulkan_INCLUDE_DIR "${AILO_DEPENDENCY_ROOT}/vulkan-headers/install/include" CACHE PATH "Pinned Vulkan headers" FORCE)
  set(Vulkan_LIBRARY "${AILO_DEPENDENCY_ROOT}/vulkan-loader/install/lib/libvulkan.so" CACHE FILEPATH "Pinned Vulkan loader" FORCE)
  set(SPIRV-Headers_DIR "${AILO_DEPENDENCY_ROOT}/spirv-headers/install/share/cmake/SPIRV-Headers")
  find_package(SPIRV-Headers CONFIG REQUIRED)
endif()
add_subdirectory("${AILO_DEPENDENCY_BUILD_ROOT}/llama/source" "${AILO_DEPENDENCY_BUILD_ROOT}/llama/build" EXCLUDE_FROM_ALL)
if(AILO_VULKAN)
  target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
endif()

if(AILO_BUILD_TESTS)
  add_executable(ailo_backend_test tests/backend.cpp)
  target_compile_options(ailo_backend_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(ailo_backend_test PRIVATE -Wl,--fatal-warnings)
  target_link_libraries(ailo_backend_test PRIVATE llama ggml mtmd)
  if(AILO_VULKAN)
    target_compile_definitions(ailo_backend_test PRIVATE AILO_TEST_VULKAN)
  endif()
  ailo_local_executable(ailo_backend_test)
  add_test(NAME backend COMMAND ailo_backend_test)
  set_tests_properties(backend PROPERTIES TIMEOUT 60 LABELS "unit;local")
  add_executable(ailo_c89_policy tests/c89_policy.c)
  ailo_c89_target(ailo_c89_policy)
  ailo_local_executable(ailo_c89_policy)
  target_link_options(ailo_c89_policy PRIVATE -Wl,--fatal-warnings)
  add_test(NAME c89-policy COMMAND ailo_c89_policy)
  set_tests_properties(c89-policy PROPERTIES TIMEOUT 10 LABELS "unit;contract")
  add_test(NAME upstream-inventory COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/upstream_inventory.py" --source "${AILO_DEPENDENCY_BUILD_ROOT}/llama/source")
  set_tests_properties(upstream-inventory PROPERTIES TIMEOUT 60 LABELS "local;contract")
  add_test(NAME archive-cache COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_archive_cache.py")
  set_tests_properties(archive-cache PROPERTIES TIMEOUT 30 LABELS "unit;local;contract")
  if(NOT CMAKE_CROSSCOMPILING AND AILO_TARGET_ID STREQUAL "x86_64-linux-gnu" AND AILO_VULKAN AND CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_test(NAME lifecycle COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_lifecycle.py")
    set_tests_properties(lifecycle PROPERTIES TIMEOUT 60 LABELS "local;contract")
  endif()
endif()
